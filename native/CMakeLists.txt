cmake_minimum_required(VERSION 3.22)
project(BayEcosystem LANGUAGES CXX)
include(FetchContent)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
FetchContent_Declare(godot_cpp GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git GIT_TAG e83fd0904c13356ed1d4c3d09f8bb9132bdc6b77)
FetchContent_MakeAvailable(godot_cpp)
add_library(bay_ecosystem SHARED src/ecosystem.cpp src/binding.cpp)
target_link_libraries(bay_ecosystem PRIVATE godot-cpp)
target_compile_features(bay_ecosystem PRIVATE cxx_std_17)
target_compile_definitions(bay_ecosystem PRIVATE NOMINMAX)
set_target_properties(bay_ecosystem PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/native/$<0:unused>" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/native/$<0:unused>")
if(MSVC)
 target_compile_options(bay_ecosystem PRIVATE /O2 /W4)
endif()
